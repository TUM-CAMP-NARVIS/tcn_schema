# Tests of the generated message code (TCN_SCHEMA_BUILD_MSG_TESTS=ON):
#   roundtrip  every struct and union of every package survives encode -> decode (generated from types.json)
#   golden     fixed values encode to exactly the bytes in test/golden/*.hex
#   msg/*      hand-written round trips of PersonTracking, HumanPose and TimeRef

# One generated source per package of the package table.
set(roundtrip_sources)
set(roundtrip_libs)
set(roundtrip_decls "")
set(roundtrip_calls "")
string(JSON n LENGTH "${TCN_PACKAGE_TABLE_JSON}" packages)
math(EXPR last "${n} - 1")
foreach(i RANGE ${last})
  string(JSON family GET "${TCN_PACKAGE_TABLE_JSON}" packages ${i} family)
  tcn_package_built(${i} built)  # ROS 2 tiers that are not selected have no library to test
  if(built AND (family STREQUAL "ros2" OR family STREQUAL "tcn"))
    string(JSON pkg GET "${TCN_PACKAGE_TABLE_JSON}" packages ${i} name)
    set(out "${CMAKE_CURRENT_BINARY_DIR}/roundtrip_${pkg}.cpp")
    add_custom_command(
            OUTPUT "${out}"
            COMMAND "${Python3_EXECUTABLE}" -I -B "${PROJECT_SOURCE_DIR}/tools/idl_cli.py" roundtrip
                    --types "${TCN_TABLES_DIR}/types.json" --package ${pkg} --out "${out}"
            DEPENDS "${TCN_TABLES_DIR}/types.json" "${PROJECT_SOURCE_DIR}/tools/idl/roundtrip.py"
            COMMENT "Generating the round-trip test of ${pkg}"
            VERBATIM)
    list(APPEND roundtrip_sources "${out}")
    list(APPEND roundtrip_libs tcn_schema_${pkg})
    string(APPEND roundtrip_decls "void roundtrip_${pkg}();\n")
    string(APPEND roundtrip_calls "    roundtrip_${pkg}();\n")
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/roundtrip_main.cpp.in"
     "// Generated by test/CMakeLists.txt\n#include \"check.h\"\n${roundtrip_decls}\nint main()\n{\n${roundtrip_calls}    return tcn_test_exit(\"roundtrip\");\n}\n")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/roundtrip_main.cpp.in" "${CMAKE_CURRENT_BINARY_DIR}/roundtrip_main.cpp" COPYONLY)

add_executable(tcn_schema_roundtrip_tests "${CMAKE_CURRENT_BINARY_DIR}/roundtrip_main.cpp" ${roundtrip_sources})
add_dependencies(tcn_schema_roundtrip_tests tcn_schema_tables)
target_include_directories(tcn_schema_roundtrip_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tcn_schema_roundtrip_tests PRIVATE ${roundtrip_libs} ${DDS_LIBRARIES})
set_target_properties(tcn_schema_roundtrip_tests PROPERTIES CXX_STANDARD 17)
add_test(NAME roundtrip COMMAND tcn_schema_roundtrip_tests)
